# 一、算法定位

算法 E 是“最优路径回放算法”。

它的逻辑是：

```text
1. 根据输入 data.maze 识别当前是哪一张已知地图。
2. 找到该地图对应的最优路径 path。
3. 按 path 一步一步移动。
4. 路上正常结算金币、陷阱、BOSS、出口。
5. 遇到 BOSS 时，不用贪心技能，而是使用预设最优技能序列。
6. 最终输出 success、move_steps、final_coin、coin_step_ratio 等字段。
```

该算法允许使用输入 `data.maze` 来识别地图，因为它本质上是 benchmark path replay，不是未知地图探索 AI。

如果你的系统仍然需要展示 fog 动画，也可以在回放过程中继续调用 `revealFog`，但路径选择本身不依赖 fogMap。

# 六、主函数 simulateOptimalPathReplay

实现：

```javascript
function simulateOptimalPathReplay(data) {
  const mazeId = detectKnownMaze(data);

  if (mazeId === "unknown") {
    return {
      success: false,
      path: [],
      path_length: 0,
      move_steps: 0,
      final_coin: 0,
      coin_step_ratio: 0,
      boss_success: false,
      boss_total_turns: 0,
      boss_revive_count: 0,
      boss_coin_cost: 0,
      boss_skill_sequence_lengths: [],
      boss_skill_sequences: [],
      collectedCoins: [],
      triggeredTraps: [],
      algorithm: "E_OPTIMAL_PATH_REPLAY",
      error: "Unknown maze. No replay path available."
    };
  }

  const plan = OPTIMAL_REPLAY_TABLE[mazeId];

  const groundTruth = data.maze.map(row => row.slice());
  const fogMap = data.maze.map(row => row.map(() => null));

  let coins = 0;
  let success = false;
  let bossSuccess = false;
  let bossBattleUsed = false;

  const rawPath = [];
  const collectedCoins = [];
  const triggeredTraps = [];

  let bossTotalTurns = 0;
  let bossReviveCount = 0;
  let bossCoinCost = 0;
  let bossSkillSequenceLengths = [];
  let bossSkillSequences = [];

  for (let i = 0; i < plan.path.length; i++) {
    const pos = plan.path[i];
    const [r, c] = pos;

    rawPath.push([...pos]);

    revealFog(fogMap, groundTruth, pos, 1);

    const cell = groundTruth[r][c];

    if (cell === "C" || cell === "G") {
      coins += 50;
      collectedCoins.push([...pos]);
      groundTruth[r][c] = " ";
      fogMap[r][c] = " ";
    }

    else if (cell === "T") {
      coins -= 30;
      triggeredTraps.push([...pos]);
      groundTruth[r][c] = " ";
      fogMap[r][c] = " ";
    }

    else if (cell === "B") {
      const battle = runBossBattleReplay(
        data.B,
        data.PlayerSkills,
        data.minRouds,
        data.CoinConsumption,
        coins,
        plan.bossSkillSequence
      );

      bossBattleUsed = true;
      bossSuccess = battle.won;
      coins = battle.coinsAfter;
      bossTotalTurns = battle.totalTurns;
      bossReviveCount = battle.reviveCount;
      bossCoinCost = battle.coinCost;
      bossSkillSequenceLengths = battle.skillSequenceLengths;
      bossSkillSequences = battle.skillSequences;

      if (!battle.won) {
        success = false;
        break;
      }

      groundTruth[r][c] = " ";
      fogMap[r][c] = " ";
    }

    else if (cell === "E") {
      if (bossSuccess || !containsBoss(data.maze)) {
        success = true;
        break;
      }
    }
  }

  const dedupedPath = dedupePath(rawPath);

  const path_length = dedupedPath.length;
  const move_steps = Math.max(0, path_length - 1);
  const final_coin = coins;
  const coin_step_ratio =
    success && move_steps > 0 ? final_coin / move_steps : 0;

  return {
    success,
    path: dedupedPath,
    path_length,
    move_steps,
    final_coin,
    coin_step_ratio,

    boss_success: bossSuccess,
    boss_total_turns: bossTotalTurns,
    boss_revive_count: bossReviveCount,
    boss_coin_cost: bossCoinCost,
    boss_skill_sequence_lengths: bossSkillSequenceLengths,
    boss_skill_sequences: bossSkillSequences,

    collectedCoins,
    triggeredTraps,

    algorithm: "E_OPTIMAL_PATH_REPLAY",
    maze_id: mazeId
  };
}
```

# 七、BOSS 回放函数

算法 E 不需要每次重新搜索技能序列。
它直接使用路径表里的最优技能序列。

实现：

```javascript
function runBossBattleReplay(
  bossHpList,
  rawSkills,
  minRounds,
  coinConsumption,
  startingCoins,
  skillSequence
) {
  let bossIndex = 0;
  let currentHp = bossHpList[0];
  let cooldowns = rawSkills.map(() => 0);

  const usedSequence = [];

  for (let turn = 0; turn < skillSequence.length; turn++) {
    const skillIndex = skillSequence[turn];
    const skill = rawSkills[skillIndex];

    if (cooldowns[skillIndex] !== 0) {
      return {
        won: false,
        coinsAfter: startingCoins,
        totalTurns: turn,
        reviveCount: 0,
        coinCost: 0,
        skillSequenceLengths: [usedSequence.length],
        skillSequences: [usedSequence],
        error: "Skill used while cooling down"
      };
    }

    currentHp -= skill[0];
    usedSequence.push(skillIndex);

    if (currentHp <= 0) {
      bossIndex++;

      if (bossIndex >= bossHpList.length) {
        return {
          won: true,
          coinsAfter: startingCoins,
          totalTurns: usedSequence.length,
          reviveCount: 0,
          coinCost: 0,
          skillSequenceLengths: [usedSequence.length],
          skillSequences: [usedSequence]
        };
      }

      currentHp = bossHpList[bossIndex];
    }

    cooldowns = updateCooldownsAfterUse(cooldowns, rawSkills, skillIndex);
  }

  return {
    won: false,
    coinsAfter: startingCoins,
    totalTurns: usedSequence.length,
    reviveCount: 0,
    coinCost: 0,
    skillSequenceLengths: [usedSequence.length],
    skillSequences: [usedSequence],
    error: "Replay skill sequence did not defeat all bosses"
  };
}
```

冷却更新规则：

```javascript
function updateCooldownsAfterUse(cooldowns, rawSkills, usedSkillIndex) {
  const next = cooldowns.map((cd, i) => {
    if (i === usedSkillIndex) {
      return rawSkills[i][1];
    }
    return Math.max(0, cd - 1);
  });

  return next;
}
```

注意：这里采用的冷却规则是：

```text
使用的技能设置为自身 cooldown
其他技能 cooldown - 1，最低为 0
```

如果你项目里的冷却规则是“先全部 -1，再给使用技能设置 cooldown”，请确保路径表中的技能序列按你的项目规则验证。如果不一致，以项目现有 `runBossBattle` 的冷却规则为准，并同步调整 `updateCooldownsAfterUse`。

# 八、工具函数

```javascript
function findCell(maze, target) {
  for (let r = 0; r < maze.length; r++) {
    for (let c = 0; c < maze[0].length; c++) {
      if (maze[r][c] === target) {
        return [r, c];
      }
    }
  }
  return null;
}

function findCells(maze, target) {
  const result = [];

  for (let r = 0; r < maze.length; r++) {
    for (let c = 0; c < maze[0].length; c++) {
      if (maze[r][c] === target) {
        result.push([r, c]);
      }
    }
  }

  return result;
}

function findCoins(maze) {
  const result = [];

  for (let r = 0; r < maze.length; r++) {
    for (let c = 0; c < maze[0].length; c++) {
      if (maze[r][c] === "C" || maze[r][c] === "G") {
        result.push([r, c]);
      }
    }
  }

  return result;
}

function samePos(a, b) {
  return (
    a &&
    b &&
    a[0] === b[0] &&
    a[1] === b[1]
  );
}

function containsBoss(maze) {
  return findCell(maze, "B") !== null;
}

function revealFog(fogMap, groundTruth, pos, viewRadius) {
  const [r, c] = pos;

  for (let dr = -viewRadius; dr <= viewRadius; dr++) {
    for (let dc = -viewRadius; dc <= viewRadius; dc++) {
      const nr = r + dr;
      const nc = c + dc;

      if (
        nr >= 0 &&
        nr < groundTruth.length &&
        nc >= 0 &&
        nc < groundTruth[0].length
      ) {
        fogMap[nr][nc] = groundTruth[nr][nc];
      }
    }
  }
}

function dedupePath(rawPath) {
  const out = [];

  for (let i = 0; i < rawPath.length; i++) {
    if (
      i === 0 ||
      rawPath[i][0] !== out[out.length - 1][0] ||
      rawPath[i][1] !== out[out.length - 1][1]
    ) {
      out.push(rawPath[i]);
    }
  }

  return out;
}
```

# 九、路径合法性校验

请实现一个可选函数，用于调试路径是否有效：

```javascript
function validateReplayPath(data, path) {
  const maze = data.maze;

  for (let i = 1; i < path.length; i++) {
    const prev = path[i - 1];
    const cur = path[i];

    const manhattan =
      Math.abs(prev[0] - cur[0]) + Math.abs(prev[1] - cur[1]);

    if (manhattan !== 1) {
      return {
        ok: false,
        error: "Non-adjacent move",
        index: i,
        prev,
        cur
      };
    }

    const [r, c] = cur;

    if (
      r < 0 ||
      r >= maze.length ||
      c < 0 ||
      c >= maze[0].length
    ) {
      return {
        ok: false,
        error: "Out of bounds",
        index: i,
        cur
      };
    }

    if (maze[r][c] === "#") {
      return {
        ok: false,
        error: "Hit wall",
        index: i,
        cur
      };
    }
  }

  return { ok: true };
}
```

在 `simulateOptimalPathReplay` 开始时可以调用：

```javascript
const check = validateReplayPath(data, plan.path);
if (!check.ok) {
  return {

    path_length: 0,
    move_steps: 0,
    final_coin: 0,
    coin_step_ratio: 0,
    boss_success: false,
    boss_total_turns: 0,
    boss_revive_count: 0,
    boss_coin_cost: 0,
    boss_skill_sequence_lengths: [],
    boss_skill_sequences: [],
    collectedCoins: [],
    triggeredTraps: [],
    algorithm: "E_OPTIMAL_PATH_REPLAY",
    maze_id: mazeId,
    error: check.error,

  };
}
```
